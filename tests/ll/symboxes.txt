; Checks for the standalone SymBoxes unit test (build/symboxes).

; CHECK: === SymExpr: constants ===
; CHECK: 0 == 0: true
; CHECK: 0 == 1: false
; CHECK: 0 + 5 == 5: true
; CHECK: 5 + 0 == 5: true
; CHECK: 5 - 0 == 5: true
; CHECK: 5 - 5 == 0: true
; CHECK: 0 * 5 == 0: true
; CHECK: 1 * 5 == 5: true
; CHECK: 5 / 1 == 5: true
; CHECK: 5 / 0 == 0: true

; CHECK: === SymExpr: infinity ===
; CHECK: +∞ + 1 == +∞: true
; CHECK: -∞ - 1 == -∞: true
; CHECK: +∞ / -1 == -∞: true
; CHECK: -∞ / -1 == +∞: true

; CHECK: === SymExpr: APInt constants and overflow ===
; CHECK: INT64_MAX + 1 prints as: 9223372036854775808
; CHECK: i128 -1 prints as: -1
; CHECK: MAX + 1 == +∞: true
; CHECK: MIN - 1 == -∞: true
; CHECK: MAX * 2 == +∞: true
; CHECK: MAX * -2 == -∞: true
; CHECK: MIN / -1 == +∞: true
; CHECK: min(-1, 2) == -1: true
; CHECK: max(-1, 2) == 2: true

; CHECK: === SymExpr: min/max ===
; CHECK: +∞ == 0: false
; CHECK: -∞ == 0: false
; CHECK: min(a,b) == max(a,b): false
; CHECK: min(a,b) == min(b,a): true
; CHECK: min(a,b)+1 == 1: false
; CHECK: min(a,a) prints as: a
; CHECK: min(3,b) prints as: min(3, b)
; CHECK: max(3,b) prints as: max(3, b)
; CHECK: [3,3] ⊔ [b,b] = [min(3, b), max(3, b)]
; CHECK: min(a,b)+1 metadata: min(1+a, 1+b)
; CHECK: (min(a,b)+1)*2 metadata: min(2+2*a, 2+2*b)

; CHECK: === SymExpr: canonical form ===
; CHECK: min(a, a+1) = a
; CHECK: max(a+3, a, a+1) = 3+a
; CHECK: min(min(a,b),c) = min(a, b, c)
; CHECK: min(a, max(a,b)) = a
; CHECK: -min(a,b) = max(-a, -b)
; CHECK: (4a+8)/4 = 2+a
; CHECK: a/4 = idiv(a,4)
; CHECK: a/4 - a/4 == 0: true
; CHECK: -7/2 = -3
; CHECK: a*b sign unknown times min = top: true
; CHECK: min of 9 symbols is top: true
; CHECK: as a range: [-∞, +∞]
; CHECK: eval x+1 = [-127, 128]
; CHECK: eval min(a+1, 5) = [1, 5]
; CHECK: eval max(a, b) = [0, +inf]
; CHECK: eval a*x = [-1280, 1270]
; CHECK: eval x/4 = [-32, 31]
; CHECK: [0,+inf] narrow [0,200] = [0, 200]
; CHECK: [0,5] narrow [0,3] = [0, 3]

; CHECK: === SymExpr: equality ===
; CHECK: v == v: true
; CHECK: v == s: false
; CHECK: v+s == s+v: true
; CHECK: v+s == v+v: false
; CHECK: v*s == s*v: true
; CHECK: (v+s)-s == v: true

; CHECK: === SymExpr: distributivity ===
; CHECK: (a+b)*c == a*c+b*c: true
; CHECK: (a-b)*c == a*c-b*c: true

; CHECK: === SymExpr: print ===
; CHECK: x+1 prints as: 1+x
; CHECK: +∞ prints as: +∞
; CHECK: -∞ prints as: -∞

; CHECK: === SymRange: basic ===
; CHECK: [0,10] prints as: [0, 10]
; CHECK: ∅ prints as: ∅
; CHECK: full prints as: [-∞, +∞]
; CHECK: [0,10] == [0,10]: true
; CHECK: ∅ == ∅: true
; CHECK: [0,10] == ∅: false

; CHECK: === SymRange: join ===
; CHECK: [0,5] ⊔ [1,10] = [0, 10]
; CHECK: ∅ ⊔ [0,5] = [0, 5]
; CHECK: [0,5] ⊔ ∅ = [0, 5]
; CHECK: [0,10] ⊔ [-∞,+∞] = [-∞, +∞]
; CHECK: [-∞,+∞] ⊔ [0,10] = [-∞, +∞]
; CHECK: ∅ ⊔ ∅ = ∅

; CHECK: === SymRange: widen ===
; CHECK: [0,5] ∇ [0,10] = [0, +∞]
; CHECK: [1,5] ∇ [0,5] = [-∞, 5]
; CHECK: [0,5] ∇ [1,10] = [-∞, +∞]
; CHECK: [0,5] ∇ [0,5] = [0, 5]
; CHECK: ∅ ∇ [0,5] = [0, 5]
; CHECK: [0,5] ∇ ∅ = [0, 5]

; CHECK: === SymRange: symbolic bounds ===
; CHECK: [0,N] prints as: [0, N]
; CHECK: [0,N+1] prints as: [0, 1+N]
; CHECK: [0,N] == [0,N]: true
; CHECK: [0,N] == [0,N+1]: false
; CHECK: [0,N] ⊔ [0,N+1] prints as: [0, 1+N]
